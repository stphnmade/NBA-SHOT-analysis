NoteInjury Note For Interpretation

Stephen Curry broke his left hand on October 30, 2019 (season 2019-20), underwent surgery, and missed 58 consecutive games.
He played only 5 games that season, briefly returned in March 2020, and Golden State finished 15-50.
Trend visuals mark 2019-20 so you can see the break in form and what happened around that season.

playerYear = await d3.csv("data/player_year.csv", d3.autoType)
madeShots = await d3.csv("data/made_shots.csv", d3.autoType)

typeaheadInput = (options, {label, value, placeholder, maxSuggestions = 5}) => {
  const listId = `suggest-${Math.random().toString(36).slice(2, 10)}`
  const wrapper = html`<label style="display:grid; gap:0.35rem;">
    <span>${label} (search)</span>
    <input type="text" list="${listId}" value="${value ?? ""}" placeholder="${placeholder ?? ""}" style="width:100%;">
    <datalist id="${listId}"></datalist>
    <small style="opacity:0.75;">Top ${maxSuggestions} suggestions while typing</small>
  </label>`
  const input = wrapper.querySelector("input")
  const datalist = wrapper.querySelector("datalist")
  const optionStrings = options.map(option => String(option))

  const refreshSuggestions = () => {
    const query = input.value.trim().toLowerCase()
    const matches = optionStrings
      .filter(option => option.toLowerCase().includes(query))
      .slice(0, maxSuggestions)
    datalist.replaceChildren(...matches.map(match => html`<option value="${match}"></option>`))
  }

  const emitInput = () => {
    wrapper.value = input.value
    wrapper.dispatchEvent(new Event("input", {bubbles: true}))
  }

  input.addEventListener("input", () => {
    refreshSuggestions()
    emitInput()
  })
  input.addEventListener("change", emitInput)

  wrapper.value = input.value
  refreshSuggestions()
  return wrapper
}

adaptiveInput = (options, config) => (
  options.length > 10
    ? typeaheadInput(options, config)
    : Inputs.select(options, {label: config.label, value: config.value})
)

resolveSearchValue = (inputValue, options, fallback) => {
  if (options.length === 0) return fallback ?? null
  const fallbackValue = options.includes(fallback) ? fallback : options[0]
  const query = String(inputValue ?? "").trim().toLowerCase()
  if (!query) return fallbackValue
  const exact = options.find(option => String(option).toLowerCase() === query)
  if (exact) return exact
  const partial = options.find(option => String(option).toLowerCase().includes(query))
  return partial ?? fallbackValue
}

  playerYear
    .filter(d => d.player_name === selectedPlayer)
    .map(d => d.team_name)
    .filter(Boolean)
)).sort((a, b) => d3.ascending(a, b))]

  value: "All",
  placeholder: "Type team name..."
})

teamFilter = resolveSearchValue(teamFilterInput, playerTeams, "All")

inWindow = d => d.season >= startSeason && d.season <= endSeason

playerRows = playerYear.filter(d => d.player_name === selectedPlayer && inWindow(d) && withTeam(d))
shotRows = madeShots.filter(d => d.player_name === selectedPlayer && inWindow(d) && withTeam(d))

summarize = rows => {
  const fga = d3.sum(rows, d => d.fga)
  const fgm = d3.sum(rows, d => d.fgm)
  const fga3 = d3.sum(rows, d => d.fga3)
  const fgm3 = d3.sum(rows, d => d.fgm3)
  const games = d3.sum(rows, d => d.games)
  return {
    fg_pct: fga > 0 ? fgm / fga : NaN,
    tp_pct: fga3 > 0 ? fgm3 / fga3 : NaN,
    efg: fga > 0 ? (fgm + 0.5 * fgm3) / fga : NaN,
    tpa_per_game: games > 0 ? fga3 / games : NaN,
    tpa_per_season: rows.length > 0 ? fga3 / rows.length : NaN,
    fga_total: fga,
    fga3_total: fga3
  }
}

agg = summarize(playerRows)
fmtPct = v => Number.isFinite(v) ? d3.format(".1%")(v) : "n/a"
fmtNum = v => Number.isFinite(v) ? d3.format(".2f")(v) : "n/a"
fmtInt = v => Number.isFinite(v) ? d3.format(",.0f")(v) : "n/a"

arcPoints = (cx, cy, r, startDeg, endDeg, steps = 80) => {
  const pts = []
  const start = startDeg * Math.PI / 180
  const end = endDeg * Math.PI / 180
  for (let i = 0; i <= steps; i++) {
    const t = start + (end - start) * (i / steps)
    pts.push({x: cx + r * Math.cos(t), y: cy + r * Math.sin(t)})
  }
  return pts
}

courtSegments = () => {
  const segs = []
  const pushLine = (a, b, id) => segs.push({id, points: [{x: a[0], y: a[1]}, {x: b[0], y: b[1]}]})

  pushLine([-25, 0], [25, 0], "baseline")
  pushLine([-25, 0], [-25, 47], "left_sideline")
  pushLine([25, 0], [25, 47], "right_sideline")
  pushLine([-25, 47], [25, 47], "halfcourt")

  pushLine([-8, 0], [-8, 19], "paint_left")
  pushLine([8, 0], [8, 19], "paint_right")
  pushLine([-8, 19], [8, 19], "paint_top")

  segs.push({id: "restricted", points: arcPoints(0, 5.25, 4, 0, 180, 70)})
  segs.push({id: "freethrow", points: arcPoints(0, 19, 6, 0, 360, 120)})
  pushLine([-22, 0], [-22, 14], "corner3_left")
  pushLine([22, 0], [22, 14], "corner3_right")
  segs.push({id: "arc3", points: arcPoints(0, 5.25, 23.75, 22, 158, 120)})

  return segs
}
